    1. Create a `mergeRecords` utility function (can be in same file or a separate `src/core/data/mergeRecords.ts`):
       ```typescript
       interface MergeableRecord {
         id: string;
         updatedAt: number;
         [key: string]: unknown;
       }

       function mergeRecords(
         existing: MergeableRecord[],
         incoming: MergeableRecord[],
       ): { merged: MergeableRecord[]; summary: { updated: number; inserted: number; unchanged: number } } {
         const existingMap = new Map(existing.map(r => [r.id, r]));
         let updated = 0, inserted = 0, unchanged = 0;

         for (const record of incoming) {
           const current = existingMap.get(record.id);
           if (!current) {
             existingMap.set(record.id, record);
             inserted++;
           } else if (record.updatedAt > current.updatedAt) {
             existingMap.set(record.id, record);
             updated++;
           } else {
             unchanged++;
           }
         }

         return {
           merged: Array.from(existingMap.values()),
           summary: { updated, inserted, unchanged },
         };
       }
       ```

       - Write merged records back via WriteJournal for atomicity
       - Show merge summary Alert: `message.success('Import complete. ${updated} records updated, ${inserted} new, ${unchanged} unchanged.')`

    3. Import wrap: `writeJournal.begin('import-data', ...)` — if WriteJournalEntry type doesn't include 'import-data', add it.
