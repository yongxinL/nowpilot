      conversationHistory: Array<{ role: 'user' | 'assistant'; content: string }>;

      availableTools: Array<{ name: string; description: string; inputSchema: Record<string, unknown> }>;
      personaBehavior: { brevity: string; clarificationStrategy: string; reasoningStyle: string } | null;

    }

      | { type: 'text-delta'; content: string }
      | { type: 'tool-call'; toolName: string; input: unknown }

      | { type: 'error'; error: import('./PipelineError').PipelineError }
      | { type: 'done'; usage?: { promptTokens: number; completionTokens: number } };

    // Tool execution

      name: string;
      description: string;
      inputSchema: Record<string, unknown>;
      execute: (input: unknown, signal?: AbortSignal) => Promise<unknown>;
    }

    export interface ToolExecutionResult {
      toolName: string;
      output: unknown;
      durationMs: number;
    }

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| User input → AgentOrchestrator | Prompt text from user crosses into the pipeline; untrusted input. |
| AgentOrchestrator → ProviderRouter → AI SDK → External AI provider | API keys (from encrypted ApiKeyStore) and user prompts sent to external services. |
| AI provider → ProviderRouter → StreamAdapter | Response text/streaming data from external provider enters the extension. |
| Pipeline → ApiKeyStore | Decrypted API keys retrieved in-memory; must never reach logs/error messages/output. |

## STRIDE Threat Register

| Threat ID | Category | Component | Severity | Disposition | Mitigation Plan |
|-----------|----------|-----------|----------|-------------|-----------------|
| T-03-01 | Spoofing | ExecutorService | high | mitigate | ExecutorService validates toolName against a closed `z.enum()` derived from registered tools. Unknown/hallucinated names → `NO_SUCH_TOOL` PipelineError. SDK's `inputSchema` validates tool inputs before `execute()`. |
| T-03-02 | Tampering | ExecutorService | medium | mitigate | AI SDK's `inputSchema` validates tool inputs against Zod schemas before execution. Invalid inputs throw `InvalidToolInputError`. |
| T-03-03 | Information Disclosure | ProviderRouter | high | mitigate | ProviderRouter reads keys from ApiKeyStore (encrypted AES-GCM-256). Keys are never included in error messages, streaming output, or PipelineError.diagnostic. Secret redaction applied before any diagnostic logging. |
| T-03-04 | Denial of Service | AgentOrchestrator | medium | mitigate | `abortSignal` from PlannerContext is forwarded to `generateText`/`streamText`. AI SDK auto-propagates to tool `execute()`. Verify with integration test: create AbortController, call abort(), assert no further tokens generated (per Pitfall 3). |
| T-03-05 | Elevation of Privilege | PersonaInjector (seed in this plan, full in Plan 03-05) | medium | mitigate | PersonaInjector prepends persona block AFTER the [SYSTEM] section header. User input is in [USER INPUT: current turn] section — clearly separated from system instructions. No user-controlled content enters the system prompt section. |
| T-03-06 | Information Disclosure | ProviderRouter | low | accept | Error messages returned to user are generic ("Authentication failed for {provider}. Check your API key in Options."). Detailed diagnostic metadata is logged to console only (consumed by Phase 6 DiagnosticsStore). Accept low risk that console logs could be inspected. |
| T-03-SC | Tampering | npm install (ai, @ai-sdk/*) | high | mitigate | All 4 @ai-sdk/* packages flagged [SUS] by RESEARCH.md Package Legitimacy Audit — latest version published 2026-07-29 (less than 24h ago at planning time). These are Vercel official packages with millions of weekly downloads; the [SUS] verdict is a publish-timestamp artifact, not a quality signal. Verified on npm registry. The `zod@4` package name was removed as [SLOP]. Executor verifies: `pnpm ls --depth=0` confirms pinned versions match RESEARCH.md. If install fails, halt with checkpoint:human-verify. |
</threat_model>
