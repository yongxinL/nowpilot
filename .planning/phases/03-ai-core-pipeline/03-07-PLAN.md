    ```typescript
    async runTurn(
      providerId: PipelineProviderId,
      tier: ModelTier,
      context: PlannerContext,
    ): Promise<string> {
      // 1. Get tier caps
      const caps = TierCapForTier(tier);
      
      // 2. Select provider via ProviderRouter (includes fallback)
      const { adapter, providerId: selectedProvider } = await providerRouter.selectProvider(providerId);
      
      // 3. Loop state
      let stepCount = 0;
      let toolCallHistory: ToolCallRecord[] = [];
      let lastDecision: PlannerDecision | null = null;
      
      // 4. Main loop — planner-controlled termination per D-05
      while (stepCount < caps.planner) {
        stepCount++;
        
        // Assemble PlannerContext with accumulated tool history
        const stepContext = { ...context, toolCallHistory };
        
        // Inject planner persona (behavioral only per D-09)
        const plannerSystem = PersonaInjector.inject('planner', buildPlannerSystem(), { profile });
        
        // Call PlannerService
        let decision: PlannerDecision;
        try {
          decision = await PlannerService.plan(adapter, tier, stepContext);
        } catch (error) {
          // Dispatch error per D-11
          return dispatchError(error, selectedProvider, context);
        }
        
        lastDecision = decision;
        
        // Route based on decision (per D-05):
        switch (decision.action) {
          case 'answer':
          case 'ask_clarification':
            // Terminal states → render
            return renderResponse(adapter, tier, decision, context);
          
          case 'run_tool':
            // Non-terminal → execute tool → loop
            const result = await ExecutorService.execute(
              decision.toolName,
              decision.input,
              context.availableTools,
              context.abortSignal,
            );
            toolCallHistory.push({
              toolName: decision.toolName,
              input: decision.input,
              output: result.output,
              timestamp: Date.now(),
            });
            // Continue loop
            break;
        }
      }
      
      // 5. Tier cap reached per D-06 — force synthesize
      return rendererService.synthesize(adapter, tier, { 
        action: 'answer', 
        reasonCode: 'tier_cap_reached' 
      }, context);
    }
    ```

    ```typescript
    function dispatchError(error: unknown, providerId: PipelineProviderId, context: PlannerContext): string {
      if (error instanceof PipelineError) {
        if (error.retryable) {
          // Try fallback via ProviderRouter
          try {
            return providerRouter.executeWithFallback(providerId, async (adapter, pid) => {
              // Re-run the turn with fallback provider
              // Simplified: just return the error message for now
              return error.userFacingMessage;
            });
          } catch (fallbackError) {
            // All fallbacks failed
            return (fallbackError instanceof PipelineError)
              ? fallbackError.userFacingMessage
              : 'All providers are temporarily unavailable. Please try again later.';
          }
        } else {
          // Terminal error → surface to user
          return error.userFacingMessage;
        }
      }
      // Unexpected error
      return 'An unexpected error occurred. Please try again.';
    }
    ```
