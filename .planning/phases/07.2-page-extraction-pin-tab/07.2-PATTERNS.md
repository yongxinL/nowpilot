```typescript
private extractContent(doc: Document): { markdown?: string; extractionType: string; extractionQuality: string } {
  try {
    if (isProbablyReaderable(doc)) {
      const clone = doc.cloneNode(true) as Document; // D-26/MUST: clone before Readability
      const article = new Readability(clone).parse();
      if (article?.content && (article.textContent?.length ?? 0) >= 100) {
```

```typescript
// Envelope wrapping pattern — all cross-context messages use RuntimeEnvelope
import type { PageContext } from './PageContext';

export class PageContextBridge {
  /**
   * Serialize current page state to PageContext and send to Background SW.
   * Follows the stateless chrome.runtime.sendMessage pattern (D-02).
   */
  async sendPageContextUpdate(pageContext: PageContext): Promise<void> {
    const envelope = {
      type: 'PAGE_CONTEXT_UPDATED',
      source: 'content-script' as const, // NEW source type added to runtimeEnvelope.ts
      payload: pageContext,
      timestamp: Date.now(),
    };

    try {
      await chrome.runtime.sendMessage(envelope);
      debugLog('debug', '[PageContextBridge] Page context update sent', {
        url: pageContext.url,
        extractionType: pageContext.extractionType,
      });
    } catch (err) {
      debugLog('error', '[PageContextBridge] Failed to send page context', { error: err });
      throw err; // Let caller handle — extraction still succeeds, messaging failure is separate
    }
  }
}
```

**Edge cases:**
- Send failures should NOT crash the extraction pipeline — catch, log, and return the extracted context anyway.
- If SW is terminated (MV3), `sendMessage` will re-awaken it — no special handling needed.
- Timestamp must be set on the envelope for ordering (D-03: latest PageContext wins).

### 4. `src/core/content/SPANavigationWatcher.ts` (utility, event-driven)

**Analog:** `src/entrypoints/background.ts` (chrome.runtime.onMessage listener pattern) + `src/components/sidepanel/SidepanelRoot.tsx` (ResizeObserver pattern lines 36-48)

**Observer/event listener pattern** (SidepanelRoot.tsx lines 36-48, background.ts lines 6-9, 16-37):
```typescript
import { debugLog } from '../utils/debugLog';

type NavigationCallback = (url: string, title: string) => void;

export class SPANavigationWatcher {
  private callback: NavigationCallback | null = null;
  private debounceTimer: ReturnType<typeof setTimeout> | null = null;
  private readonly DEBOUNCE_MS = 300; // D-24

  // Attach/detach pattern from SidepanelRoot.tsx lines 36-48 (ResizeObserver)
  watch(callback: NavigationCallback): () => void {
    this.callback = callback;

    // Monkey-patch History API (Pitfall 4: save originals, wrap, delegate)
    const origPushState = history.pushState.bind(history);
    const origReplaceState = history.replaceState.bind(history);
    const self = this;

    history.pushState = function(...args: any[]) {
      origPushState.apply(history, args);
      self.onNavigationChange(location.href, document.title);
    };
    history.replaceState = function(...args: any[]) {
      origReplaceState.apply(history, args);
      self.onNavigationChange(location.href, document.title);
    };

    const popstateHandler = () => this.onNavigationChange(location.href, document.title);
    window.addEventListener('popstate', popstateHandler);
    window.addEventListener('hashchange', popstateHandler);

    // Title MutationObserver for SPAs that update <title> after navigation
    const titleObserver = new MutationObserver(() => {
      this.onNavigationChange(location.href, document.title);
    });
    const titleEl = document.querySelector('title');
    if (titleEl) {
      titleObserver.observe(titleEl, { childList: true, characterData: true, subtree: true });
    }

    // Return cleanup function (same pattern as SidepanelRoot line 47)
    return () => {
      this.callback = null;
      history.pushState = origPushState;
      history.replaceState = origReplaceState;
      window.removeEventListener('popstate', popstateHandler);
      window.removeEventListener('hashchange', popstateHandler);
      titleObserver.disconnect();
      if (this.debounceTimer) clearTimeout(this.debounceTimer);
    };
  }

  // Debounce pattern (D-24: 300ms)
  private onNavigationChange(url: string, title: string): void {
    if (!this.callback) return;
    if (this.debounceTimer) clearTimeout(this.debounceTimer);
    this.debounceTimer = setTimeout(() => {
      debugLog('debug', '[SPANavigationWatcher] Navigation detected', { url });
      this.callback?.(url, title);
    }, this.DEBOUNCE_MS);
  }

  // Public: destroy cleans up
  destroy(): void {
    if (this.debounceTimer) clearTimeout(this.debounceTimer);
    this.callback = null;
  }
}
```

**Edge cases:**
- `runAt: 'document_start'` in WXT config is critical (Pitfall 4): monkey-patch History API before page scripts load.
- Save references to original methods before wrapping — call original first, then dispatch custom logic.
- Title observer: some SPAs update `<title>` after URL change; catch both signals.
- 300ms debounce allows routing transitions and content rendering to settle (D-24).

```typescript
// src/entrypoints/content.ts
import { defineContentScript } from 'wxt/utils/define-background';
import { PageExtractor } from '../core/content/PageExtractor';
import { PageContextBridge } from '../core/content/PageContextBridge';
import { SPANavigationWatcher } from '../core/content/SPANavigationWatcher';
import { ContentChangeWatcher } from '../core/content/ContentChangeWatcher';
import { validateEnvelope } from '../core/messaging/runtimeEnvelope';
import { debugLog } from '../core/utils/debugLog';

export default defineContentScript({
  matches: ['<all_urls>'],
  runAt: 'document_idle',
  world: 'ISOLATED', // D-05: ISOLATED only for v7.2
  main(ctx: ContentScriptContext) {
    // --- Module composition (D-01: single entrypoint, four modules) ---
    const extractor = new PageExtractor();
    const bridge = new PageContextBridge();
    const navWatcher = new SPANavigationWatcher();
    const changeWatcher = new ContentChangeWatcher(); // dormant

    // --- URL blocklist check (D-26) ---
    if (!shouldExtract(location.href)) {
      debugLog('info', '[content.ts] Skipping extraction for blocked URL', { url: location.href });
      return;
    }

    // --- Extract & send helper ---
    async function extractAndSend(): Promise<void> {
      try {
        const pageContext = extractor.extract(document);
        await bridge.sendPageContextUpdate(pageContext);
      } catch (err) {
        debugLog('error', '[content.ts] Extraction failed', { error: err });
      }
    }

    // --- Trigger 1: DOMContentLoaded (initial injection) ---
    if (document.readyState !== 'loading') {
      extractAndSend(); // DOM already parsed (document_idle)
    } else {
      document.addEventListener('DOMContentLoaded', extractAndSend, { once: true });
    }

    // --- Trigger 2: SPA navigation (D-03, D-23) ---
    navWatcher.watch((_url, _title) => {
      extractAndSend();
    });

    // --- Trigger 3: Tab visibility change (D-25: hidden → visible) ---
    document.addEventListener('visibilitychange', () => {
      if (document.visibilityState === 'visible') {
        extractAndSend();
      }
    });

    // --- Message handler: explicit extraction request from SW (D-03) ---
    // Pattern from background.ts lines 16-37 (chrome.runtime.onMessage)
    chrome.runtime.onMessage.addListener((message, _sender, sendResponse) => {
      if (message && typeof message === 'object' && message.type === 'GET_PAGE_CONTEXT_REQUEST') {
        extractAndSend().then(() => {
          // Re-read latest PageContext from workspaceStore via SW relay
          // For now, respond with the extracted context directly
          const pageContext = extractor.extract(document);
          sendResponse({ success: true, pageContext });
        }).catch(err => {
          sendResponse({ success: false, error: err instanceof Error ? err.message : String(err) });
        });
        return true; // async response (background.ts line 33 pattern)
      }
    });

    // --- Cleanup on content script invalidation (WXT lifecycle) ---
    ctx.onInvalidated(() => {
      navWatcher.destroy();
      changeWatcher.destroy();
    });
  },
});

// URL blocklist helper (D-26)
function shouldExtract(url: string): boolean {
  const blocked = /^(chrome|chrome-extension|edge|about|view-source|devtools|file):\/\//i;
  if (blocked.test(url)) return false;
  // Block NowPilot-owned pages
  if (url.startsWith(`chrome-extension://${chrome.runtime.id}`)) return false;
  return true;
}
```

**Edge cases:**
- `document.readyState` check: if already 'interactive'/'complete', DOM exists and extraction can run immediately; otherwise wait for `DOMContentLoaded`.
- `chrome.runtime.onMessage` must return `true` for async responses (same as background.ts line 33).
- WXT `ctx.onInvalidated()` for cleanup when content script is unloaded.
- URL blocklist check must happen FIRST before any extraction or observation (security requirement D-26).

```typescript
    return new Promise((resolve) => {

          resolve({ success: false, error: 'No active tab' });
          return;
        }
        chrome.tabs.sendMessage(tab.id, { type: 'GET_PAGE_CONTEXT_REQUEST' }, (response) => {
          if (chrome.runtime.lastError) {
            resolve({ success: false, error: chrome.runtime.lastError.message });
          } else {
            resolve(response || { success: false, error: 'No response from content script' });
          }
        });
      });
    });
  },
```

### 8. `src/core/ai/tools/builtin/pinTabTool.ts` (tool, request-response)

```typescript
import { z } from 'zod';
import type { ToolDefinition } from '../ToolDefinition';
import { useWorkspaceStore } from '../../stores/workspaceStore';

export const pinTabTool: ToolDefinition = {
  name: 'pin-tab',
  description: 'Pin or unpin a tab for context reference. Maximum 10 pinned tabs.',
  // Zod v4 input schema (D-21)
  inputSchema: z.object({
    tabId: z.number().describe('The browser tab ID to pin or unpin'),
    action: z.enum(['pin', 'unpin']).default('pin').describe('Pin or unpin the tab'),
  }),
  outputSchema: z.object({
    success: z.boolean(),
    pinnedCount: z.number().optional(),
    error: z.string().optional(),
  }),
  category: 'safe',
  requiresPermission: false,

  async execute(input, context) {
    if (context.abortSignal.aborted) throw new DOMException('Aborted', 'AbortError');
    const { tabId, action } = input as { tabId: number; action: 'pin' | 'unpin' };
    const store = useWorkspaceStore.getState();

    if (action === 'unpin') {
      store.removePinnedTab(tabId);
      return { success: true, pinnedCount: store.pinnedTabs.length - 1 };
    }

    // D-30: Max 10, reject at limit
    if (store.pinnedTabs.length >= 10) {
      return { success: false, error: 'Maximum 10 pinned tabs reached. Unpin a tab before pinning a new one.' };
    }

    // Get tab metadata via chrome.tabs (SW context required)
    try {
      const tab = await chrome.tabs.get(tabId);
      const tabContext = {
        tabId: tab.id!,
        windowId: tab.windowId,
        page: store.currentPageContext || {
          url: tab.url || '', title: tab.title || '', origin: '', hostname: '',
          meta: {}, extractedAt: Date.now(),
          extractionType: 'metadata-only', extractionQuality: 'minimal',
        },
        pinnedAt: Date.now(),
        active: true,
        url: tab.url,
        title: tab.title,
      };
      store.addPinnedTab(tabContext);
      return { success: true, pinnedCount: store.pinnedTabs.length };
    } catch (err) {
      return { success: false, error: `Could not pin tab: ${err instanceof Error ? err.message : String(err)}` };
    }
  },
};
```

```typescript
  const visibleTabs = pinnedTabs.slice(0, 5);
  const overflowCount = Math.max(0, pinnedTabs.length - 5);

  return (
    <div style={containerStyle} data-testid="pin-tab-bar">
      {/* Current page chip — always visible (D-12) */}
      {currentPageContext ? (
        <Tag
          color="blue"
          icon={isCurrentPinned ? <PushpinOutlined /> : undefined}
          closable={isCurrentPinned}
          onClose={() => removePinnedTab(/* current tabId */)}
          style={{ maxWidth: 160, overflow: 'hidden', textOverflow: 'ellipsis' }}
        >
          {currentPageContext.title?.slice(0, 20) || currentPageContext.url}
        </Tag>
      ) : null}

      {/* Pinned tab chips */}
      {visibleTabs.map((tab) => (
        <Tag
          key={tab.tabId}
          color={tab.active ? 'default' : 'default'}
          closable
          onClose={() => removePinnedTab(tab.tabId)}
          icon={tab.active ? undefined : <CloseOutlined />}
          style={{ maxWidth: 160, overflow: 'hidden', textOverflow: 'ellipsis' }}
        >
          {tab.title?.slice(0, 20) || tab.url || `Tab ${tab.tabId}`}
        </Tag>
      ))}

      {/* Overflow count + management popover (D-11) */}
      {overflowCount > 0 ? (
        <Popover
          open={popoverOpen}
          onOpenChange={setPopoverOpen}
          content={
            <div style={{ maxWidth: 300 }}>
              {pinnedTabs.map((tab) => (
                <div key={tab.tabId} style={{ display: 'flex', justifyContent: 'space-between', padding: '4px 0' }}>
                  <span>{tab.title || tab.url}</span>
                  <Button type="text" size="small" icon={<CloseOutlined />}
                    onClick={() => removePinnedTab(tab.tabId)} />
                </div>
              ))}
            </div>
          }
          trigger="click"
        >
          <Tag>+{overflowCount} more</Tag>
        </Popover>
      ) : null}
    </div>
  );
}
```

**Edge cases:**
- Zustand selectors (individual field selectors) — prevents re-renders when unrelated workspace state changes (Pitfall 5).
- `token` from `theme.useToken()` for consistent antd styling (SidepanelContent line 25).
- Max 10 pins enforced by store.setter (D-30) — UI doesn't need to re-validate.

```typescript
  const fields: string[] = [];
  if (context.title) fields.push(`Title: ${context.title}`);
  if (context.url) fields.push(`URL: ${context.url}`);
  if (context.markdown) {
    // D-31: include first ~200 chars of markdown for pinned tab summaries
    const prefix = String(context.markdown).slice(0, 200).trim();
    fields.push(`Content: ${prefix}${String(context.markdown).length > 200 ? '...' : ''}`);
  }

  if (fields.length === 0) {
    const str = JSON.stringify(context);
    return this.heuristicTruncate(str, 300);
  }
  const result = fields.join('\n');
  const estimated = tokenEstimator.estimateTokens(result);

  return result;
```
