```typescript
async analyze(...): Promise<...> {
  await providerRegistry.initialize();
  // tier → costTier mapping
  const models = providerRegistry.getModelsForTier('haiku');
  if (models.length === 0) return fallback;
  
  const provider = providerRegistry.getProvider(models[0].providerId);
  if (!provider) return fallback;
  const model = (provider.instance as any)(models[0].modelId);
  
  for (let attempt = 1; attempt <= 2; attempt++) {
    try {
      const { text } = await generateText({ model, system: PROMPT, prompt, maxTokens: 500 });
      // Parse + validate with repairAndValidate (NOT manual JSON.parse)
      const parsed = repairAndValidate(text, schema, fallback);
      if ('result' in parsed) return parsed.result;
      if (attempt === 2) return parsed.fallback;
    } catch (err) {
      if (attempt === 2) { debugLog('error', '[...] failed after retry', { error: err }); return fallback; }
      debugLog('warn', '[...] retrying');
    }
  }
}
```

import React, { useEffect, useState } from 'react';
import { Form, Typography, Button, Switch, App } from 'antd';
const { Title } = Typography;
const STORAGE_KEY = 'np_notes_llm_features';

export function NotesSection() {
  const { message } = App.useApp();
  const [features, setFeatures] = useState<LlmFeatureToggles>({...});
  const [loading, setLoading] = useState(false);

  useEffect(() => {
    loadFeatures();
  }, []);

  const loadFeatures = async () => {
    try {
      const result = await chrome.storage.local.get(STORAGE_KEY);
      setFeatures({ autoTag: true, ..., ...result[STORAGE_KEY] });

      await chrome.storage.local.set({ [STORAGE_KEY]: features });

      message.error('Failed to save');
    } finally { setLoading(false); }
  };

  return (
    <div data-options-section="notes" style={{ maxWidth: 720 }}>
      <Title level={4}>Notes</Title>
      <p style={{ marginBottom: 16 }}>LLM-powered note features and backup configuration.</p>

      <Button type="primary" onClick={handleSave} loading={loading}>Save</Button>
