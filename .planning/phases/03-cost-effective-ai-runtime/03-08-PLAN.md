  <action>
    Create `PromptCacheManager` class+singleton.

    **Class structure:**
    - Private `#cacheKeys = new Map<string, CacheKey>()` — per-provider cache keys
    - Private `#sectionHints = new Map<string, CacheHint[]>()` — cached section identification results

    **`identifyStableSections(promptParts: Array<{ role: string; content: string; section?: CacheSection }>): Map<number, CacheHint>`:**
    - Iterates `promptParts` — for each part with a `section` tag (system-prompt, tool-schemas, preferences, memory), creates a `CacheHint` with that part's index
    - Returns `Map<number, CacheHint>` — message index → cache hint (provider-agnostic)
    - Pure logic — no side effects, no persistence

    **`generateCacheKey(providerId: string): string`:**
    - If key exists for providerId and not expired → return existing hash
    - If not → generate new key: `hash = simpleHash(providerId + Date.now().toString())`, store in `#cacheKeys`
    - Returns the hash string

    **`invalidateCacheKey(providerId: string, reason: string): void`:**
    - Removes cache key for providerId
    - Logs via `debugLog('info', '[PromptCacheManager] Cache key invalidated', { providerId, reason })`
    - Called when: provider config changes, tool schemas change, system prompts change, memory facts change per D-15

    **`invalidateAll(): void`:**
    - Clears all cache keys
    - Logs via `debugLog('info', '[PromptCacheManager] All cache keys invalidated')`

    **`simpleHash(input: string): string`:** Internal helper — use a simple DJB2 or similar hash (NOT crypto.subtle — this is for cache keys, not security). `let hash = 5381; for (let i = 0; i < input.length; i++) { hash = ((hash << 5) + hash) + input.charCodeAt(i); } return hash.toString(36);`

    **Export:** `export const promptCacheManager = new PromptCacheManager()` singleton + `export class PromptCacheManager`.

    **Test file:** `tests/core/ai/cache/PromptCacheManager.test.ts` — test section identification, cache key generation, invalidation, deterministic hash output.

    **Per-provider mapping (D-16):**

    ```typescript
    // Anthropic: per-message cacheControl on marked messages
    export function applyAnthropicCache(
      messages: Array<{ role: string; content: string | Array<{ type: string; text?: string }> }>,
      hintMap: Map<number, CacheHint>
    ): Array<{ role: string; content: unknown; providerOptions?: unknown }> {
      return messages.map((msg, idx) => {
        if (!hintMap.has(idx)) return msg;
        return {
          ...msg,
          providerOptions: {
            anthropic: { cacheControl: { type: 'ephemeral' as const } },
          },
        };
      });
    }
    ```

    ```typescript
    // OpenAI: promptCacheKey on system message, promptCacheBreakpoint marker
    export function applyOpenAICache(
      messages: Array<{ role: string; content: string }>,
      hintMap: Map<number, CacheHint>,
      cacheKey: string
    ): { messages: Array<{ role: string; content: string }>; providerOptions: Record<string, unknown> } {
      // OpenAI uses request-level providerOptions, not per-message
      const providerOptions: Record<string, unknown> = {};
      const hasCachedSections = hintMap.size > 0;

      if (hasCachedSections) {
        providerOptions.openai = {
          promptCacheKey: cacheKey,
          promptCacheOptions: { mode: 'auto', ttl: 3600 },
        };
      }

      // Mark last cached message as breakpoint
      const messagesWithBreakpoints = messages.map((msg, idx) => {
        if (hintMap.has(idx)) {
          return { ...msg, providerOptions: { openai: { promptCacheBreakpoint: true } } };
        }
        return msg;
      });

      return { messages: messagesWithBreakpoints, providerOptions };
    }
    ```

    ```typescript
    // Gemini: cachedContent for the system prompt
    export function applyGoogleCache(
      messages: Array<{ role: string; content: string }>,
      hintMap: Map<number, CacheHint>
    ): Array<{ role: string; content: unknown; providerOptions?: unknown }> {
      return messages.map((msg, idx) => {
        if (!hintMap.has(idx)) return msg;
        return {
          ...msg,
          providerOptions: {
            google: { cachedContent: msg.content },
          },
        };
      });
    }
    ```

    ```typescript
    // Dispatcher: routes to correct adapter based on provider type
    export function applyCacheHints(
      providerType: string,
      messages: Array<{ role: string; content: string }>,
      hintMap: Map<number, CacheHint>,
      cacheKey: string
    ): { messages: Array<{ role: string; content: unknown }>; providerOptions?: Record<string, unknown> } {
      switch (providerType) {
        case 'anthropic':
          return { messages: applyAnthropicCache(messages, hintMap) };
        case 'openai':
          return applyOpenAICache(messages, hintMap, cacheKey);
        case 'google':
          return { messages: applyGoogleCache(messages, hintMap) };
        case 'ollama':
        default:
          // No cache support — return messages unchanged
          return { messages };
      }
    }
    ```
