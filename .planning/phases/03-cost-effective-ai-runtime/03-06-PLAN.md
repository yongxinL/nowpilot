    **StructuredOutput** (`src/core/ai/pipeline/StructuredOutput.ts`) — pure function, no class:

    ```typescript
    import { jsonrepair, JSONRepairError } from 'jsonrepair';
    import { z } from 'zod';
    import { debugLog } from '../../utils/debugLog';

    export interface RepairResult<T> {
      result?: T;
      fallback?: T;
    }

    export function repairAndValidate<T>(
      text: string,
      schema: z.ZodType<T>,
      fallback: T
    ): { result: T } | { fallback: T } {
      // 1. One-shot JSON repair
      let repaired: string;
      try {
        repaired = jsonrepair(text);
      } catch (err) {
        if (err instanceof JSONRepairError) {
          debugLog('warn', '[StructuredOutput] JSON repair failed', { position: err.position });
          return { fallback };
        }
        throw err;
      }

      // 2. Parse + validate
      try {
        const parsed = JSON.parse(repaired);
        const result = schema.safeParse(parsed);
        if (result.success) return { result: result.data };
      } catch (err) {
        debugLog('warn', '[StructuredOutput] JSON parse failed', { error: err });
      }

      // 3. Schema validation failed → fallback
      debugLog('warn', '[StructuredOutput] Schema validation failed — returning fallback');
      return { fallback };
    }
    ```

    **Constructor DI:** `constructor(private toolRegistry: ToolRegistry, private permissionService: PermissionService)`

    **`async execute(toolName: string, toolInput: Record<string, unknown>, abortSignal: AbortSignal): Promise<ToolExecutionResult>`:**

    1. **Closed-enum validation:** `toolRegistry.get(toolName)` → if undefined, return structured error `{ success: false, error: 'Unknown tool: {toolName}' }`
    2. **Permission check:** `await permissionService.canExecute(toolName, toolInput)` → if false, return `{ success: false, error: 'Permission denied for tool: {toolName}' }`
    3. **Input validation:** `tool.inputSchema.safeParse(toolInput)` — Zod v4 → if fail, return error with Zod error message
    4. **Execution:** `await tool.execute(validatedInput, { abortSignal })` in try/catch:
       - AbortError → `{ success: false, error: 'Tool execution timed out' }`
       - Other errors → `{ success: false, error: err.message }`
    5. **Output validation:** `tool.outputSchema.safeParse(output)` — if fail, return error
    6. **Success:** `{ success: true, output: validatedOutput }`

    All errors logged via `debugLog('error', ...)`. Never throws — always returns structured `ToolExecutionResult`.
